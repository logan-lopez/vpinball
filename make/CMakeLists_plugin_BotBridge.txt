# plugin CMake file to be included in main project CMakelists

# --- BotBridge plugin ---

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 99)

set(BOTBRIDGE_PLUGIN_SOURCES
   plugins/bot-bridge/botbridge.cpp
)

if(BUILD_SHARED)
   add_library(BotBridgePlugin MODULE ${BOTBRIDGE_PLUGIN_SOURCES})

   target_include_directories(BotBridgePlugin PUBLIC
      ${CMAKE_SOURCE_DIR}/third-party/include
      plugins
      plugins/bot-bridge
   )

   if(PluginPlatform STREQUAL "windows" OR PluginPlatform STREQUAL "windows-mingw")
      if(MSVC)
         add_compile_options(
            $<$<CONFIG:RELEASE>:/Ob2>
            $<$<CONFIG:RELEASE>:/O2>
            $<$<CONFIG:RELEASE>:/Oi>
            $<$<CONFIG:RELEASE>:/arch:SSE2>
            $<$<CONFIG:RELEASE>:/fp:fast>
            $<$<CONFIG:RELEASE>:/fp:except->
            $<$<CONFIG:RELEASE>:/Ot>
            $<$<CONFIG:RELEASE>:/GF>
            $<$<CONFIG:RELEASE>:/GS->
            $<$<CONFIG:RELEASE>:/Gy>
            $<$<CONFIG:RELEASE>:/GR>
            $<$<CONFIG:RELEASE>:/Oy>
            $<$<CONFIG:RELEASE>:/GT>
            $<$<CONFIG:RELEASE>:/GL>
         )
         set_target_properties(BotBridgePlugin PROPERTIES
            MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:DEBUG>:Debug>"
         )
         target_compile_options(BotBridgePlugin PUBLIC
            $<$<CONFIG:RELEASE>:$<$<COMPILE_LANGUAGE:CXX>:${OPT_COMMON}>>
            $<$<CONFIG:RELEASE>:$<$<COMPILE_LANGUAGE:C>:${OPT_COMMON}>>
         )
         target_link_options(BotBridgePlugin PUBLIC
            $<$<CONFIG:RELEASE>:/INCREMENTAL:NO>
            $<$<CONFIG:RELEASE>:/OPT:REF>
            $<$<CONFIG:RELEASE>:/OPT:ICF>
            $<$<CONFIG:RELEASE>:/LTCG>
         )
      endif()
      target_link_libraries(BotBridgePlugin ws2_32)
      if (PluginArch STREQUAL "x64")
         set_target_properties(BotBridgePlugin PROPERTIES
            PREFIX ""
            OUTPUT_NAME "plugin-bot-bridge64"
         )
      else()
         set_target_properties(BotBridgePlugin PROPERTIES
            PREFIX ""
            OUTPUT_NAME "plugin-bot-bridge"
         )
      endif()
   else()
      set_target_properties(BotBridgePlugin PROPERTIES
         PREFIX ""
         OUTPUT_NAME "plugin-bot-bridge"
      )
      if(PluginPlatform STREQUAL "macos")
         set_target_properties(BotBridgePlugin PROPERTIES
            SUFFIX ".dylib"
         )
      endif()
   endif()

   add_custom_command(TARGET BotBridgePlugin POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy "${CMAKE_SOURCE_DIR}/plugins/bot-bridge/plugin.cfg" "${PLUGINS_DIR}/bot-bridge/plugin.cfg"
      COMMAND "${CMAKE_COMMAND}" -E copy "$<TARGET_FILE:BotBridgePlugin>" "${PLUGINS_DIR}/bot-bridge"
      COMMAND "${CMAKE_COMMAND}" -E rm "$<TARGET_FILE:BotBridgePlugin>"
   )
endif()

if(BUILD_STATIC)
   add_compile_options(-fPIC)

   add_library(BotBridgePlugin STATIC ${BOTBRIDGE_PLUGIN_SOURCES})

   target_include_directories(BotBridgePlugin PUBLIC
      ${CMAKE_SOURCE_DIR}/third-party/include
      plugins
      plugins/bot-bridge
   )

   set_target_properties(BotBridgePlugin PROPERTIES
      PREFIX ""
      OUTPUT_NAME "plugin-bot-bridge"
   )
endif()
